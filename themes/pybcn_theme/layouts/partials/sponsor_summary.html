{{- /* One sponsor logo, linked to the sponsor site. Logos come from assets/images/sponsors/ and go through the Hugo image pipeline. The grid caps a logo at 150 px high, so a 560 px WebP covers the widest slot, with a 1120 px variant for a 2x screen. Three cases stay unprocessed: an external logo URL, because Hugo cannot process a remote file; an SVG, which needs no raster step; and a GIF, because processing would flatten an animated logo to one frame. */ -}} {{- $src := $.sponsor.Params.logo -}} {{- $srcset := "" -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- if not $src -}} {{- with resources.Get (printf "images/sponsors/%s" $.sponsor.Params.logo_image) -}} {{- $subtype := .MediaType.SubType -}} {{- if or (eq $subtype "svg") (eq $subtype "gif") -}} {{- $src = .RelPermalink -}} {{- if ne $subtype "svg" -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- end -}} {{- else -}} {{- $small := .Fit "560x150 webp q80" -}} {{- $large := .Fit "1120x300 webp q80" -}} {{- $src = $small.RelPermalink -}} {{- $width = $small.Width -}} {{- $height = $small.Height -}} {{- if gt $large.Width $small.Width -}} {{- $srcset = printf "%s 1x, %s 2x" $small.RelPermalink $large.RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
